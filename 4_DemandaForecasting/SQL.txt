CREATE DATABASE ds4;



/* Creando tablas*/
CREATE TABLE categorias(
id_categoria SERIAL PRIMARY KEY,
categoria TEXT,
descripcion TEXT);

CREATE TABLE clientes(
id_cliente SERIAL PRIMARY KEY,
nombre TEXT,
apellido TEXT,
email TEXT,
fecha TEXT,
region TEXT);

CREATE TABLE metodos(
id_metodo SERIAL PRIMARY KEY,
metodo TEXT,
descripcion TEXT);

CREATE TABLE productos(
id_producto SERIAL PRIMARY KEY,
nombre TEXT,
categoria TEXT,
precio SMALLINT,
stock SMALLINT);

CREATE TABLE ventas(
id_venta SERIAL PRIMARY KEY,
fecha TEXT,
cliente_id INT,
FOREIGN KEY (cliente_id) REFERENCES clientes(id_cliente),
producto_id INT,
FOREIGN KEY (producto_id) REFERENCES productos(id_producto),
cantidad SMALLINT,
metodo_id INT,
FOREIGN KEY (metodo_id) REFERENCES metodos(id_metodo),
estado TEXT);

/* Creando tabla necesaria*/
CREATE TABLE ventas_productos AS
SELECT
ventas.id_venta,
ventas.fecha,
ventas.cliente_id,
ventas.cantidad,
productos.stock,
productos.nombre,
productos.categoria,
productos.precio
FROM ventas
INNER JOIN productos
ON ventas.producto_id = productos.id_producto
ORDER BY ventas.id_venta;
